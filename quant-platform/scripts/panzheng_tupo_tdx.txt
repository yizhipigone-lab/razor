{ Eurica Quant 睿奕量化 — 盘整突破策略 }
{ 通达信选股公式 }

N:=5;
ZF:=8;

{══════════ Y1: 阳包阴——今天收阳昨天收阴 ══════════}
Y1:=C>O AND REF(C<O,1);

{══════════ 上一次Y1到现在的K线数 ══════════}
BARS_Y1:=BARSLAST(Y1);

{══════════ Y1当天的最低价和最高价 ══════════}
Y1_LOW:=REF(L,BARS_Y1);
Y1_HIGH:=REF(H,BARS_Y1);

{══════════ 从Y1至今的最低价 ══════════}
LOW_PRICE:=LLV(L,BARS_Y1+1);

{══════════ 横盘条件: Y1后N天内, 最低价从未跌破Y1当天最低价 ══════════}
HENGPAN:=BARS_Y1>0 AND BARS_Y1<=N AND LOW_PRICE>=Y1_LOW;

{══════════ XG4: 最近15天内至少出现过一次5连阳 ══════════}
XG4:=COUNT(EVERY(C>O,5),15)>=1;

{══════════ 突破: 今天阳线 + 涨幅>=ZF% + 突破Y1最高价 + 横盘中 + 5连阳背景 ══════════}
BREAK1:=C>O AND (C/REF(C,1)-1)*100>=ZF AND C>Y1_HIGH AND HENGPAN AND XG4;

{══════════ 信号（5天内去重）══════════}
BREAK:=FILTER(BREAK1,5);

{══════════ 选股输出 ══════════}
BREAK;
