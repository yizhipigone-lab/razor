```javascript
let currentReportPage = 1;
const REPORT_PAGE_SIZE = 15;

async function loadReportsList() {
  const searchInput = document.getElementById('report-search');
  const search = searchInput ? searchInput.value.trim() : '';
  const offset = (currentReportPage - 1) * REPORT_PAGE_SIZE;
  const listEl = document.getElementById('report-list');
  if (!listEl) return;

  listEl.innerHTML = '<p style="color:var(--text2);text-align:center;padding:20px">加载中...</p>';

  try {
    const r = await fetch(
      `/api/reports/list?limit=${REPORT_PAGE_SIZE}&offset=${offset}&search=${encodeURIComponent(search)}`
    ).then(resp => resp.json());

    if (r.status === 'ok' && r.data) {
      const { total, data } = r.data;

      if (!data || data.length === 0) {
        listEl.innerHTML = '<p style="color:var(--text2);text-align:center;padding:20px">暂无历史报告</p>';
      } else {
        listEl.innerHTML = data.map(rep => `
          <div class="strategy-item" onclick="viewReport(${rep.id})" style="cursor:pointer">
            <div style="display:flex;justify-content:space-between;align-items:center">
              <span style="font-weight:600;font-size:13px">${rep.code} ${rep.name || ''}</span>
              <button class="btn btn-ghost btn-sm" style="padding:2px 6px;font-size:10px"
                onclick="event.stopPropagation();deleteReport(${rep.id},'${rep.code}')">🗑️</button>
            </div>
            <div style="font-size:11px;color:var(--text2);margin-top:2px">${rep.created_at || ''}</div>
          </div>
        `).join('');
      }

      // 更新分页信息
      const totalPages = Math.max(1, Math.ceil(total / REPORT_PAGE_SIZE));
      const pageInfo = document.getElementById('rep-page-info');
      if (pageInfo) pageInfo.textContent = `${currentReportPage} / ${totalPages}`;
    }
  } catch (e) {
    listEl.innerHTML = '<p style="color:var(--red);text-align:center;padding:20px">加载失败</p>';
    console.error('loadReportsList failed:', e);
  }
}
```

```javascript
  const blob = new Blob([content], { type: 'text/markdown;charset=utf-8' });
  const a = document.createElement('a');
  a.href = URL.createObjectURL(blob);
  a.download = title.replace(/[^\w一-鿿]/g, '_') + '.md';

  document.body.removeChild(a);
```

### P1-1 同股票同日去重 — upsert 策略 (对应缺陷 H1)

**严重性**: 🟠 HIGH
**涉及文件**:

#### 实现步骤

**Step 1: 改造 `save_ai_report()`**

```python
def save_ai_report(self, code: str, name: str, content: str) -> int:
    """保存分析报告（同股票同日 upsert，返回报告ID）"""
    try:
        clean_code = str(code).split('.')[0]
        today = date.today().isoformat()

        # 检查今日是否已有该股票的报告
        existing = self.conn.execute(
            "SELECT id FROM ai_reports WHERE code = ? AND CAST(created_at AS DATE) = ?",
            [clean_code, today]
        ).fetchone()

        if existing:
            # 更新已有报告（刷新内容 + 时间戳）
            self.conn.execute(
                "UPDATE ai_reports SET content = ?, name = ?, created_at = NOW() WHERE id = ?",
                [content, name, existing[0]]
            )
            self.conn.commit()
            log.info(f"AI 报告已更新: code={clean_code}, id={existing[0]}")
            return existing[0]

        # 插入新报告
        result = self.conn.execute(
            "INSERT INTO ai_reports (id, code, name, content) VALUES (nextval('seq_ai_report'), ?, ?, ?) RETURNING id",
            [clean_code, name, content]
        ).fetchone()
        self.conn.commit()
        return result[0] if result else -1

    except Exception as e:
        log.error(f"Save AI report failed: {e}")
        # 序列修复重试（保留原有逻辑）
        try:
            max_id = self.conn.execute("SELECT COALESCE(MAX(id), 0) FROM ai_reports").fetchone()[0]
            self.conn.execute("DROP SEQUENCE IF EXISTS seq_ai_report")
            self.conn.execute(f"CREATE SEQUENCE seq_ai_report START {max_id + 1}")
            clean_code = str(code).split('.')[0]
            result = self.conn.execute(
                "INSERT INTO ai_reports (id, code, name, content) VALUES (nextval('seq_ai_report'), ?, ?, ?) RETURNING id",
                [clean_code, name, content]
            ).fetchone()
            self.conn.commit()
            return result[0]
        except Exception as retry_e:
            log.error(f"AI 报告自动修复重试失败: {retry_e}")
            return -1
```

```python
import time
import threading
from concurrent.futures import ThreadPoolExecutor
from typing import Dict

# 简易内存任务管理器（单进程，重启清空）
_report_tasks: Dict[str, dict] = {}
_task_lock = threading.Lock()
_executor = ThreadPoolExecutor(max_workers=2, thread_name_prefix="ai-report")

def _do_generate(task_id: str, stock_code: str, name: str):
    """后台线程执行报告生成"""
    try:
        with _task_lock:
            _report_tasks[task_id]["status"] = "generating"
            _report_tasks[task_id]["progress"] = 10

        from app.agents.committee import generate_ai_report
        report = generate_ai_report(stock_code)

        with _task_lock:
            _report_tasks[task_id]["progress"] = 80

        if not report:
            raise ValueError("AI 未能生成有效的诊断内容")

        res = db.save_ai_report(stock_code, name, report)

        with _task_lock:
            _report_tasks[task_id]["status"] = "done"
            _report_tasks[task_id]["progress"] = 100
            _report_tasks[task_id]["report"] = report
            _report_tasks[task_id]["report_id"] = res

    except Exception as e:
        import traceback
        with _task_lock:
            _report_tasks[task_id]["status"] = "error"
            _report_tasks[task_id]["error"] = str(e)
            _report_tasks[task_id]["traceback"] = traceback.format_exc()
        log.error(f"AI Report Task {task_id} failed: {e}")
```

### P3-1 报告版本管理 (对应缺陷：增强需求)

**当前问题**: 同股票不同日生成的报告互不关联，无法回溯历史。

#### 实现方案

**数据库改造**: 在 `ai_reports` 表增加 `version` 字段：

```sql
ALTER TABLE ai_reports ADD COLUMN version INTEGER DEFAULT 1;
```

**upsert 逻辑调整**: 同股票不再覆盖，而是 version+1：

```python
def save_ai_report(self, code: str, name: str, content: str) -> int:
    clean_code = str(code).split('.')[0]
    # 获取当前最大版本号
    max_ver = self.conn.execute(
        "SELECT COALESCE(MAX(version), 0) FROM ai_reports WHERE code = ?",
        [clean_code]
    ).fetchone()[0]

    result = self.conn.execute(
        """INSERT INTO ai_reports (id, code, name, content, version)
           VALUES (nextval('seq_ai_report'), ?, ?, ?, ?) RETURNING id""",
        [clean_code, name, content, max_ver + 1]
    ).fetchone()
    self.conn.commit()
    return result[0] if result else -1
```

**前端展示**: 侧边栏同股票报告折叠显示版本号，最新版默认展开。

```python
class BatchReportReq(BaseModel):
    codes: list[str]  # 股票代码列表

@router.post("/api/agents/analyze/batch")
@limiter.limit("2/minute")
def batch_analyze(request: Request, req: BatchReportReq):
    """批量生成报告（异步，逐只处理）"""
    if len(req.codes) > 20:
        return {"status": "error", "message": "单次最多20只"}
    task_ids = []
    for code in req.codes:
        if not _validate_stock_code(code):
            continue
        task_id = f"rpt_{code.split('.')[0]}_{int(time.time())}"
        with _task_lock:
            _report_tasks[task_id] = {
                "status": "queued", "progress": 0,
                "stock_code": code, "started_at": time.time()
            }
        _executor.submit(_do_generate, task_id, code, "")
        task_ids.append(task_id)
    return {"status": "started", "task_ids": task_ids, "total": len(task_ids)}
```
