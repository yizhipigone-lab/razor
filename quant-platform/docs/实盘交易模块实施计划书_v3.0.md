```python
class CircuitBreaker:
    def __init__(self, max_failures=3, timeout_sec=30):
        self._open = False
        self._open_time = None
        self._consecutive_failures = 0

    def call(self, fn, *args, **kwargs):
        if self._open:
            if time.time() - self._open_time < self._timeout:
                raise CircuitOpenError("服务熔断中")
            self._open = False
            self._consecutive_failures = 0
        try:
            result = fn(*args, **kwargs)
            self._consecutive_failures = 0
            return result
        except Exception as e:
            self._consecutive_failures += 1
            if "WaitingFreeWriter" in str(e):
                log.critical("WaitingFreeWriter 超限，强制重启")
                if WAITINGFREEWRITER_FORCE_EXIT:
                    os._exit(1)
            if self._consecutive_failures >= self._max_failures:
                self._open = True
                self._open_time = time.time()
            raise
```

```python
class ClearanceLockService:
    def __init__(self, store):
        self._store = store  # live_clearance_locks 表
        self._locks = {}  # 内存缓存
        self._order_to_key = {}  # 反查索引
        self._DEFAULT_TTL = 300  # 5分钟

    def build_key(env, account, code):
        return f"{env}::{account}::{code}"

    def acquire(self, env, account, code, reason) -> bool:
        """加锁：已存在则返回False"""
        key = self.build_key(env, account, code)
        # 1. 先清过期
        self._purge_if_expired(key)
        # 2. 已存在 → 拒绝
        if key in self._locks:
            log.warning(f"清仓锁被持有: {key}")
            return False
        # 3. 加锁 + 写库
        self._locks[key] = LockRecord(
            acquired_at=time.time(), reason=reason, ttl_sec=self._DEFAULT_TTL
        )
        self._store.upsert_lock(...)
        return True

    def is_locked(self, env, account, code) -> bool:
        """固卖规则用：锁存在则跳过该股票"""
        return self.build_key(env, account, code) in self._locks

    def bind_order_id(self, env, account, code, qmt_order_id):
        """下单成功后绑定 QMT 订单号"""
        key = self.build_key(env, account, code)
        if key in self._locks:
            self._locks[key].qmt_order_id = qmt_order_id
            self._order_to_key[qmt_order_id] = key

    def release_by_order_id(self, qmt_order_id) -> bool:
        """QMT成交/错误回调时O(1)释放"""
        key = self._order_to_key.get(qmt_order_id)
        if not key: return False
        del self._order_to_key[qmt_order_id]
        del self._locks[key]
        self._store.delete_lock(key)
        return True

    def release(self, env, account, code) -> bool:
        """异常路径释放"""
        ...

    def _purge_if_expired(self, key):
        """TTL过期自动清理"""
        if key in self._locks:
            if time.time() - self._locks[key].acquired_at > self._DEFAULT_TTL:
                del self._locks[key]
                log.warning(f"TTL过期释放: {key}")
```

```python
def prepare_sell_action(self, env, account, code, reason, orders_detail):
    """
    MQ 关键三步：上锁 → 撤单等ack → 实时刷持仓
    """
    # 1. 上锁
    if not self._lock.acquire(env, account, code, reason):
        return {"ok": False, "reason": "lock_busy"}

    try:
        # 2. 如果有在途委托 → 撤单 + 等 ack
        if orders_detail:
            cancel_ids = [o['order_id'] for o in orders_detail]
            self.cancel_batch(cancel_ids)
            # 轮询等到 ack（替代 sleep 盲等）
            self._wait_cancel_acked(cancel_ids, timeout_ms=5000)

        # 3. 实时刷新持仓（不复用 stale 缓存）
        fresh_pos = self._refresh_position(code)
        if not fresh_pos:
            self._lock.release(env, account, code)
            return {"ok": False, "reason": "no_position"}
        return {"ok": True, "fresh_pos": fresh_pos}
    except Exception as e:
        self._lock.release(env, account, code)
        raise
```

**MQ `rule-stop-loss.ts` 关键设计**（我们要照搬）：
```python
class StopLossRule:
    priority = 60
    async def evaluate(self, ctx):
        threshold = ctx.config['stop_loss_drawdown_percentage']
        if threshold <= 0: return []  # 配置未启用跳过
        
        # 强制从配置获取价格类型，禁止默认/兼容
        basis = ctx.config['stop_loss_price_type']  # '最高价' or '成本价'
        if basis not in ('最高价', '成本价'):
            raise Error('缺少或非法配置: 止损价格类型')
        
        actions = []
        for pos in ctx.positions:
            dd_high = pos.drawdown_rate
            profit_rate = pos.profit_rate
            loss_from_cost = max(0, -profit_rate)
            dd_used = loss_from_cost if basis == '成本价' else dd_high
            
            if dd_used >= threshold:
                actions.append(ExitAction(
                    stock_code=pos.stock_code,
                    target_volume=0,  # 数量在执行层算
                    trigger_type='止损',
                    priority=60,
                    note=f'止损[基准={basis}] 回撤>={threshold}%'
                ))
        return actions
```

```python
        code = str(d.stock_code)
        by_code.setdefault(code, []).append(d)
    
    for arr in by_code.values():
        arr.sort(key=lambda x: x.trade_time)
    
    result = []
    for code, arr in by_code.items():
        net = 0
        current = None
        for d in arr:
            order_type = int(d.order_type)
            # 严格只用 orderType：23=买，24=卖
            # direction 字段在股票场景下是多空方向，不能用
            is_buy = order_type == 23
            is_sell = order_type == 24
            if not (is_buy or is_sell):
                log.warning(f"无效 orderType: {d.order_type}")
                continue
            
            vol = int(d.traded_volume)
            price = float(d.traded_price)
            
            if is_buy:
                if net == 0:
                    current = SimpleCycle(status='ongoing', ...)
                net += vol
                current.buys.append({...})
            if is_sell:
                net -= vol
                current.sells.append({...})
            
            if current and net == 0:
                current.status = 'closed'
                result.append(current)
                current = None
        
        if current:
            result.append(current)  # 未闭合
    
    # 计算指标：avgCost, pnl
    for cycle in result:
        _finalize_cycle_metrics(cycle)
    return result
```

#### 5.1.3 `hold_days.py`（0.5天）

**新文件**：`app/live_trader/hold_days.py`

**照搬 MQ `hold-days-service.ts:84-187`**：

```python
def calculate_single_stock_hold_days(
    stock_code, trade_logs, trading_days
) -> Tuple[int, Optional[str]]:
    """
    基于交易日历的连续持仓天数
    """
    daily_buy, daily_sell = aggregate_daily(trade_logs)
    
    cumulative = {}
    running_total = 0
    for date in trading_days:
        buy = daily_buy.get(date, 0)
        sell = daily_sell.get(date, 0)
        running_total += buy - sell
        if running_total < 0:
            return 0, f"{date}累计持仓为负{running_total}"
        cumulative[date] = running_total
    
    # 从最近倒推连续天数
    consecutive = 0
    for i in range(len(trading_days) - 1, -1, -1):
        if cumulative.get(trading_days[i], 0) > 0:
            consecutive += 1
        else:
            break
    return consecutive, None
```

**关键配置**：`lookback_days = 配置的N + 210`（防止窗口外历史数据漏掉）

```python
class ExitRuleEngine:
    def run(self, ctx: ExitContext, mode: str = 'RUN') -> List[ExitAction]:
        """
        mode='TEST' → 只评估，不下单，返回命中股票
        mode='RUN'  → 评估 + 下单

        actions = []
        for rule in sorted(self.rules, key=lambda r: r.priority, reverse=True):
            await rule.prepare(ctx)  # 批量预查询
            actions.extend(await rule.evaluate(ctx))
        if mode == 'TEST':
            return actions  # 不下单
        # mode='RUN'：由调用方 execute_sell_actions()
        return actions
```
