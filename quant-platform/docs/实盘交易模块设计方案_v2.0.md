## 1. 总体设计原则

### 1.1 核心铁律

1. **资金安全 > 自动化 > 性能**（不可妥协）
2. **隔离优于复用**（账户/DB/进程/配置四独立）
3. **复用优于重写**（策略/风控/信号复用，成交层替换）
4. **每笔真单必可追溯**（审计全字段）
5. **可回滚**（任何时刻可一键切回监控不下单）
6. **🆕 多重保险**（紧急停止开关 + 风控门 + 用户决策保留接口）

### 1.2 不可接受的失败模式（红线）

- ❌ 1次非用户预期的真单
- ❌ 1次风控门未生效的委托
- ❌ 1次实盘数据污染模拟盘
- ❌ 1次无法紧急停止的事故

### 3.2 `app/live_trader/execution.py`

**v2.0 增补**：
- A1：启动时 + 每次发单前主动 ping（解决"卖时断连"）
- E3：所有 QMT 调用加 timeout=3 + 重试 2 次
- E4：启动时 schema 自检（xtquant 字段断言）

```python
"""
实盘成交执行层 v2.0
"""
from __future__ import annotations
import threading, time, os, sys
from typing import Optional, Dict, List
from dataclasses import dataclass
from datetime import datetime

from xtquant import xttrader, xtconstant
from xtquant.xttype import XtAccount

from core.logger import get_logger
from app.live_trader.config import (
    QMT_USERDATA_PATH, QMT_ACCOUNT_ID, QMT_SESSION_ID, DRY_RUN
)

log = get_logger("LiveExecution")


@dataclass
class OrderResult:
    order_id: int
    code: str
    direction: str
    price: float
    volume: int
    submitted_at: datetime
    status: str
    filled_volume: int = 0
    filled_avg_price: float = 0.0
    filled_amount: float = 0.0
    error_msg: str = ""


class QtQuantExecutor:
    _instance: Optional["QtQuantExecutor"] = None
    _lock = threading.Lock()

    def __init__(self):
        self._order_lock = threading.Lock()
        self._xt_trader = None
        self._account = None
        self._connected = False
        self._last_ping_ok = False
        self._last_ping_ts = None
        self._connect()

    @classmethod
    def get(cls):
        if cls._instance is None:
            with cls._lock:
                if cls._instance is None:
                    cls._instance = cls()
        return cls._instance

    def _connect(self):
        try:
            if not os.path.exists(QMT_USERDATA_PATH):
                raise FileNotFoundError(f"QMT userdata 路径不存在: {QMT_USERDATA_PATH}")
            session_id = int(time.time())
            self._xt_trader = xttrader.XtQuantTrader(QMT_USERDATA_PATH, session_id)
            self._xt_trader.start()
            res = self._xt_trader.connect()
            if res != 0:
                raise ConnectionError(f"QMT connect 失败 code={res}")
            self._account = XtAccount(QMT_ACCOUNT_ID, "STOCK")
            self._xt_trader.subscribe(self._account)
            self._connected = True
            self._last_ping_ok = True
            self._last_ping_ts = datetime.now()

            # 🆕 v2.0 E4: schema 自检
            self._schema_self_check()

            log.success(f"QMT 连接成功 account={QMT_ACCOUNT_ID} session={session_id}")
        except Exception as e:
            self._connected = False
            log.error(f"QMT 连接失败: {e}")
            raise

    # 🆕 v2.0 E4
    def _schema_self_check(self):
        """启动时确认 xtquant API 字段未变更"""
        try:
            # 抽查 query_stock_asset 返回字段
            asset = self._xt_trader.query_stock_asset(self._account)
            if asset:
                required = ["cash", "frozen_cash", "market_value", "total_asset"]
                for f in required:
                    if not hasattr(asset, f):
                        log.error(f"xtquant 字段缺失: {f}，版本可能不兼容")
                        raise RuntimeError(f"xtquant schema 不兼容: 缺 {f}")
            # xtconstant 枚举确认
            assert xtconstant.STOCK_BUY == 23, f"STOCK_BUY={xtconstant.STOCK_BUY} ≠ 23"
            assert xtconstant.STOCK_SELL == 24, f"STOCK_SELL={xtconstant.STOCK_SELL} ≠ 24"
            log.info("xtquant schema 自检通过")
        except Exception as e:
            log.error(f"schema 自检失败: {e}")
            raise

    # 🆕 v2.0 A1: 主动 ping
    def ping(self, max_age_sec: int = 30) -> bool:
        """检查 QMT 连接是否可用，缓存 30 秒"""
        now = datetime.now()
        if self._last_ping_ts and (now - self._last_ping_ts).total_seconds() < max_age_sec:
            return self._last_ping_ok
        try:
            self._xt_trader.query_stock_asset(self._account)
            self._last_ping_ok = True
        except Exception as e:
            log.warning(f"QMT ping 失败: {e}")
            self._last_ping_ok = False
        self._last_ping_ts = now
        return self._last_ping_ok

    # 🆕 v2.0 E3: 所有 QMT 调用加重试
    def _with_retry(self, fn, *args, retries=2, timeout=3, **kwargs):
        last_err = None
        for attempt in range(retries + 1):
            try:
                return fn(*args, **kwargs), None
            except Exception as e:
                last_err = e
                log.warning(f"QMT 调用失败 attempt={attempt+1} err={e}")
                time.sleep(0.3)
        return None, last_err

    def buy(self, code, price, volume, reason="") -> OrderResult:
        return self._send_order(code, price, volume, xtconstant.STOCK_BUY, "BUY", reason)

    def sell(self, code, price, volume, reason="") -> OrderResult:
        return self._send_order(code, price, volume, xtconstant.STOCK_SELL, "SELL", reason)

    def _send_order(self, code, price, volume, order_type, direction, reason) -> OrderResult:
        if not self._connected:
            raise ConnectionError("QMT 未连接")

        # 🆕 v2.0 A1: 发单前再 ping 一次
        if not self.ping(max_age_sec=5):
            return OrderResult(
                order_id=-1, code=code, direction=direction,
                price=price, volume=volume, submitted_at=datetime.now(),
                status="rejected", error_msg="QMT ping 失败",
            )

        submitted_at = datetime.now()
        if DRY_RUN:
            log.info(f"[DRY-RUN] {direction} {code} px={price:.2f} vol={volume} reason={reason}")
            return OrderResult(
                order_id=-1, code=code, direction=direction,
                price=price, volume=volume, submitted_at=submitted_at, status="dry_run",
            )

        with self._order_lock:
            try:
                fmt_code = self._format_code(code)
                order_id, err = self._with_retry(
                    self._xt_trader.order_stock,
                    self._account, fmt_code, order_type,
                    int(volume), 5, price,
                    f"P9_LIVE_{direction}", f"live_{int(time.time()*1000)}",
                )
                if err or order_id is None or order_id < 0:
                    log.error(f"{direction} 失败 {code}@{price}x{volume} err={err}")
                    return OrderResult(
                        order_id=order_id or -1, code=code, direction=direction,
                        price=price, volume=volume, submitted_at=submitted_at,
                        status="rejected", error_msg=str(err or "order_id<0"),
                    )
                log.info(f"{direction} 提交 order_id={order_id} {code}@{price}x{volume} reason={reason}")
                return OrderResult(
                    order_id=order_id, code=code, direction=direction,
                    price=price, volume=volume, submitted_at=submitted_at, status="submitted",
                )
            except Exception as e:
                log.error(f"{direction} 异常: {e}")
                return OrderResult(
                    order_id=-1, code=code, direction=direction,
                    price=price, volume=volume, submitted_at=submitted_at,
                    status="rejected", error_msg=str(e),
                )

    def cancel(self, order_id) -> bool:
        with self._order_lock:
            result, err = self._with_retry(self._xt_trader.cancel_order, self._account, order_id)
            if err:
                log.error(f"撤单失败 order_id={order_id}: {err}")
                return False
            log.info(f"撤单 order_id={order_id}")
            return True

    def query_balance(self) -> dict:
        if not self._connected: return {}
        asset, err = self._with_retry(self._xt_trader.query_stock_asset, self._account)
        if err or not asset: return {}
        return {
            "cash": float(asset.cash),
            "frozen_cash": float(asset.frozen_cash),
            "market_value": float(asset.market_value),
            "total_asset": float(asset.total_asset),
        }

    def query_positions(self) -> List[dict]:
        if not self._connected: return []
        positions, err = self._with_retry(self._xt_trader.query_stock_positions, self._account)
        if err: return []
        result = []
        for p in positions or []:
            result.append({
                "code": str(p.stock_code).split(".")[0],
                "volume": int(p.volume),
                "can_sell": int(p.can_sell_volume),
                "avg_price": float(p.open_price),
                "market_value": float(p.market_value),
            })
        return result

    def query_orders(self) -> List[dict]:
        if not self._connected: return []
        orders, err = self._with_retry(self._xt_trader.query_stock_orders, self._account)
        if err: return []
        result = []
        for o in orders or []:
            result.append({
                "order_id": int(o.order_id),
                "code": str(o.stock_code).split(".")[0],
                "direction": "BUY" if o.order_type == xtconstant.STOCK_BUY else "SELL",
                "price": float(o.price),
                "volume": int(o.order_volume),
                "filled_volume": int(o.traded_volume),
                "status": str(o.order_status),
                "submitted_at": str(o.order_time),
            })
        return result

    @staticmethod
    def _format_code(code: str) -> str:
        c = str(code).strip()
        if "." in c: return c
        if c.startswith(("5", "6", "9", "7")): return f"{c}.SH"
        if c.startswith(("0", "1", "2", "3")): return f"{c}.SZ"
        if c.startswith(("4", "8")): return f"{c}.BJ"
        return c


executor = QtQuantExecutor.get()
```

### 3.4 `app/live_trader/risk_gate.py`（风控前置门）

**v2.0 重大增补**（C1/C2/C3 修复 + 启动自检）：

```python
"""
实盘风控前置门 v2.0
- 6 道检查 + 启动合理性自检
- C1 修复：单票集中度查"已持仓+新买"
- C2 修复：可用现金 = cash - frozen_cash
- C3 修复：日亏损熔断首日 fallback
"""
from datetime import date
from typing import Tuple
from core.logger import get_logger
from app.live_trader.config import (
    MAX_SINGLE_POSITION_PCT, MAX_TOTAL_POSITION_PCT,
    MAX_DAILY_LOSS_PCT, MAX_ORDER_AMOUNT, MIN_CASH_RESERVE,
    INITIAL_CAPITAL, validate_risk_config,
)
from app.live_trader.execution import executor
from app.live_trader.store import LiveStore

log = get_logger("RiskGate")
_store = LiveStore()


class RiskDecision:
    def __init__(self, passed, reason="", metrics=None):
        self.passed = passed
        self.reason = reason
        self.metrics = metrics or {}


# 🆕 v2.0 F2: 启动时校验（main.py lifespan 调用）
def startup_self_check() -> bool:
    issues = validate_risk_config()
    if issues:
        for i in issues:
            log.error(f"[风控配置自检] {i}")
        return False
    log.info("[风控配置自检] 通过")
    return True


def check_buy(code: str, price: float, volume: int) -> RiskDecision:
    order_amount = price * volume
    metrics = {"order_amount": order_amount, "code": code}

    # 1. 单笔金额硬上限
    if order_amount > MAX_ORDER_AMOUNT:
        return _block("单笔超限", metrics, f"{order_amount:.0f} > {MAX_ORDER_AMOUNT}")

    bal = executor.query_balance()
    if not bal:
        return _block("无法获取余额", metrics, "QMT query_stock_asset 失败")

    # 🆕 v2.0 C2: 可用现金 = cash - frozen_cash
    available_cash = bal["cash"] - bal.get("frozen_cash", 0)
    if available_cash - order_amount < MIN_CASH_RESERVE:
        return _block("现金不足", metrics,
                      f"可用={available_cash:.0f} - 委托={order_amount:.0f} < 保留={MIN_CASH_RESERVE}")

    total_asset = bal["total_asset"]
    # 已有持仓（来自 QMT query_positions）
    positions = executor.query_positions()
    existing_shares = sum(p["volume"] for p in positions if p["code"] == code)
    existing_mv = sum(p["market_value"] for p in positions if p["code"] == code)

    # 🆕 v2.0 C1: 单票集中度 = (已持仓市值 + 新买金额) / total_asset
    new_total_mv = existing_mv + order_amount
    new_position_pct = new_total_mv / total_asset
    if new_position_pct > MAX_SINGLE_POSITION_PCT:
        return _block("单票超限", metrics,
                      f"持仓+新买 {new_total_mv:.0f} / 总资产 {total_asset:.0f} = {new_position_pct:.1%} > {MAX_SINGLE_POSITION_PCT:.0%}")

    # 总仓位
    current_market_value = bal["market_value"]
    total_position_pct = (current_market_value + order_amount) / total_asset
    if total_position_pct > MAX_TOTAL_POSITION_PCT:
        return _block("总仓位超限", metrics,
                      f"{total_position_pct:.1%} > {MAX_TOTAL_POSITION_PCT:.0%}")

    # 🆕 v2.0 C3: 日亏损熔断 + 首日 fallback
    today = date.today()
    today_pnl_pct = _calc_today_pnl_pct(today)
    if today_pnl_pct is None:
        # 启动首日用 INITIAL_CAPITAL 作基准
        log.info(f"[风控] 首日无昨收，用 INITIAL_CAPITAL={INITIAL_CAPITAL} 作基准")
        # 仍然通过（因为是首日没有可比基准）
    elif today_pnl_pct <= MAX_DAILY_LOSS_PCT:
        return _block("日亏损熔断", metrics,
                      f"今日={today_pnl_pct:.2%} <= {MAX_DAILY_LOSS_PCT:.2%}")

    return RiskDecision(True, "pass", metrics)


def check_sell(code, price, volume) -> RiskDecision:
    positions = executor.query_positions()
    pos = next((p for p in positions if p["code"] == code), None)
    if not pos:
        return _block("无持仓", {"code": code}, f"QMT 无 {code} 持仓")
    if volume > pos["can_sell"]:
        return _block("可卖不足", {"code": code, "want": volume, "can_sell": pos["can_sell"]})
    return RiskDecision(True, "pass", {"code": code, "volume": volume})


def _block(reason, metrics, detail) -> RiskDecision:
    log.warning(f"风控阻断: {reason} | {detail}")
    _store.write_audit("risk_blocked", metrics.get("code", ""),
                       {"level": "WARN", "reason": reason, "detail": detail, "metrics": metrics})
    try:
        from server.websocket.manager import sync_broadcast
        sync_broadcast({"type": "live_risk_block", "reason": reason, "detail": detail})
    except Exception:
        pass
    return RiskDecision(False, reason, metrics)


def _calc_today_pnl_pct(today: date) -> float:
    rows = _store.conn.execute("""
        SELECT date, equity FROM live_equity
        WHERE date IN (?, (SELECT MAX(date) FROM live_equity WHERE date < ?))
        ORDER BY date
    """, [today, today]).fetchall()
    if len(rows) < 2:
        return None
    return (rows[1][1] - rows[0][1]) / rows[0][1]
```

```python

实盘引擎 v2.0
"""
from __future__ import annotations
import sys, os
sys.path.insert(0, os.path.join(os.path.dirname(__file__), "..", ".."))

from datetime import date, datetime, timedelta
from typing import Optional, Dict, List

import threading

from app.live_trader.config import *
from app.live_trader.execution import executor, OrderResult

from app.live_trader.intraday_monitor import LiveIntradayMonitor
from core.logger import get_logger

log = get_logger("LiveEngine")

# 🆕 v2.0 C5: 全局发单串行锁
_order_serial_lock = threading.Lock()


class LiveTraderEngine:
    def __init__(self, store: LiveStore = None):
        self._store = store or LiveStore()
        self._monitor: Optional[LiveIntradayMonitor] = None

    def start(self):
        log.info("LiveTraderEngine 启动...")

        # 🆕 v2.0 F2: 启动时风控配置自检
        if not startup_self_check():
            raise RuntimeError("风控配置自检失败，拒绝启动")

        # QMT 健康检查
        bal = executor.query_balance()
        if not bal:
            raise RuntimeError("QMT 不可用")
        log.info(f"QMT 余额: total={bal['total_asset']:.0f} cash={bal['cash']:.0f}")

        # 立即对账
        self.reconcile()

        # 🆕 v2.0 A4: 启动时判断是否已过 14:53
        now = datetime.now()
        t = now.hour * 100 + now.minute
        if t > 1453:
            log.warning(f"启动时间 {now.strftime('%H:%M')} 已过 14:53，今日交易跳过，仅对账")
        else:
            if MONITOR_ENABLED:
                self._monitor = LiveIntradayMonitor(self)
                self._monitor.start()

    # 🆕 v2.0 F3: 全局 kill switch 检查
    def _check_kill_switch(self) -> bool:
        if EMERGENCY_STOP:
            log.error("[KILL-SWITCH] 紧急停止生效，跳过本阶段所有操作")
            self._store.write_audit("kill_switch_active", "", {"phase": "blocked"})
            return True
        return False

    def sell_phase(self, today, snapshot, trading_dates):
        # 🆕 F3
        if self._check_kill_switch(): return

        # 🆕 v2.0 A1: 发单前 ping
        if not executor.ping():
            log.error("QMT ping 失败，跳过卖出阶段")
            self._store.write_audit("sell_skipped", "", {"reason": "qmt_ping_fail"})
            return

        positions = self._store.load_positions()
        if not positions:
            log.info("无持仓，跳过卖出")
            return

        sells = self._check_stops(positions, snapshot, trading_dates)
        log.info(f"卖出阶段: {len(sells)} 笔待卖")

        for code, exit_price, reason, partial in sells:
            pos = positions.get(code, {})
            volume = partial if partial else pos.get("can_sell", 0)
            if volume < 100: continue

            decision = check_sell(code, exit_price, volume)
            if not decision.passed:
                self._store.write_audit("sell_blocked", code, {"reason": decision.reason})
                continue

            self._store.write_audit("sell_decision", code, {
                "price": exit_price, "volume": volume, "reason": reason,
            })

            # 🆕 v2.0 C5: 串行发单
            with _order_serial_lock:
                result = executor.sell(code, exit_price, volume, reason=reason)
                self._store.upsert_order({
                    "order_id": result.order_id, "code": code,
                    "direction": "SELL", "price": exit_price, "volume": volume,
                    "status": result.status, "reason": reason,
                    "submitted_at": result.submitted_at, "error_msg": result.error_msg,
                })

        self.reconcile()
        # 🆕 v2.0 D1: 强制 record equity
        self.record_equity(today, snapshot)

    def buy_phase(self, today, signals):
        # 🆕 F3
        if self._check_kill_switch(): return

        # A1: 发单前 ping
        if not executor.ping():
            log.error("QMT ping 失败，跳过买入")
            self._store.write_audit("buy_skipped", "", {"reason": "qmt_ping_fail"})
            return

        # 🆕 v2.0 B1: 每笔发单前 re-query balance
        bal = executor.query_balance()
        if not bal:
            log.error("无法获取余额")
            return
        available_cash = bal["cash"] - bal.get("frozen_cash", 0) - MIN_CASH_RESERVE

        bought_codes = set()
        for code, price in signals:
            if code in bought_codes: break
            if available_cash < MIN_BUY_AMT: break

            volume = int(min(MAX_ORDER_AMOUNT, available_cash) / price / 100) * 100
            if volume < 100: continue

            decision = check_buy(code, price, volume)
            if not decision.passed:
                self._store.write_audit("buy_blocked", code, {
                    "reason": decision.reason, "metrics": decision.metrics,
                })
                continue

            self._store.write_audit("buy_decision", code, {
                "price": price, "volume": volume, "amount": price * volume,
            })

            with _order_serial_lock:
                result = executor.buy(code, price, volume, reason=f"signal:{STRATEGY_NAME}")
                self._store.upsert_order({
                    "order_id": result.order_id, "code": code,
                    "direction": "BUY", "price": price, "volume": volume,
                    "status": result.status, "reason": f"signal:{STRATEGY_NAME}",
                    "submitted_at": result.submitted_at, "error_msg": result.error_msg,
                })

                if result.status != "rejected":
                    cost = price * volume * 1.0003
                    available_cash -= cost  # 🆕 B1 串行扣减
                    bought_codes.add(code)

        self.reconcile()
        self.record_equity(today, {})

    def _check_stops(self, positions, snapshot, trading_dates) -> List[tuple]:
        from app.backtest.exit_rules import exit_rule_engine
        from app.sim_trader.engine import Position

        sells = []
        for code, pos_data in positions.items():
            bar = snapshot.get(code)
            if not bar: continue
            close_p = bar['close']
            high_p = bar.get('high', close_p)

            pos = Position(
                code=code,
                entry_date=pos_data.get("entry_date", date.today()),
                entry_price=pos_data.get("entry_price", pos_data["avg_price"]),
                shares=pos_data["shares"],
                cost=pos_data.get("cost", pos_data["avg_price"] * pos_data["shares"]),
                strategy_name=pos_data.get("strategy_name", STRATEGY_NAME),
            )
            pos.peak_price = max(pos_data.get("peak_price", pos.entry_price), high_p)
            pos.remaining_shares = pos_data["can_sell"]
            pos.current_price = close_p

            sim_params = self._build_risk_params()
            ctx = exit_rule_engine.build_context(pos, bar, 1, sim_params, use_high_for_tp=True)
            signal = exit_rule_engine.check(ctx)
            if signal:
                partial = None if signal.sell_ratio >= 1.0 else int(pos.remaining_shares * signal.sell_ratio / 100) * 100
                sells.append((code, signal.sell_price, signal.reason, partial))
        return sells

    def _build_risk_params(self):
        return {
            "hard_stop": HARD_STOP,
            "take_profit_tiers": TAKE_PROFIT_TIERS,
            "trail_activate": TRAIL_ACTIVATE,
            "trail_dd": TRAIL_DD,
            "time_exit_days": TIME_EXIT_DAYS,
            "time_exit_profit": TIME_EXIT_PROFIT,
            "time_force_days": TIME_FORCE_DAYS,
            "first_day_exit_min_profit": FIRST_DAY_EXIT_MIN_PROFIT,
            "first_day_exit_days": FIRST_DAY_EXIT_DAYS,
            "use_atr_trail": USE_ATR_TRAIL,
            "atr_trail_multiplier": ATR_TRAIL_MULTIPLIER,
        }

    def reconcile(self):
        try:
            qmt_pos = executor.query_positions()
            self._store.save_positions_from_qmt(qmt_pos)
            log.debug(f"对账完成: {len(qmt_pos)} 持仓")
        except Exception as e:
            log.error(f"对账失败: {e}")
            self._store.write_audit("reconcile_failed", "", {"error": str(e)})

    def record_equity(self, today, snapshot):
        bal = executor.query_balance()
        if not bal: return
        positions = executor.query_positions()
        self._store.save_equity_point(
            today, bal["total_asset"], bal["cash"], bal["market_value"], len(positions)
        )
```

```python
def is_active() -> bool:

    if _config.EMERGENCY_STOP or val:
```
